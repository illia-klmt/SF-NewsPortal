best_auth = Author.objects.order_by('-rating').first()

best_post = Post.objects.filter(post_type='AR').order_by('-rating').first()
print(f"Дата: {best_post.created}, Автор: {best_post.author.user.username}, Рейтинг: {best_post.rating}, Заголовок: {best_post.heading}, Превью: {best_post.preview()}")

for c in Comment.objects.filter(post=best_post):
